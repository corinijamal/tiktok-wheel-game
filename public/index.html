const { TikTokLiveConnection } = require('tiktok-live-connector'); const express = require('express'); const http = require('http'); const { Server } = require('socket.io'); const path = require('path'); const app = express(); const server = http.createServer(app); // فتح منافذ الأمان بالكامل لاستقبال حزم الأزرار أونلاين const io = new Server(server, { cors: { origin: "*", methods: ["GET", "POST"] } }); // ضع هنا اسم الحساب الذي ترغب بسحب بياناته (بدون علامة @) const TARGET_USERNAME = "a_7m_dz"; let isJoinOpen = false; const joinedUsers = new Set(); // تقديم ملفات الواجهة الرسومية من مجلد public app.use(express.static(path.join(__dirname, 'public'))); let tiktokConnection = new TikTokLiveConnection(TARGET_USERNAME, { requestOptions: { timeout: 10000 }, websocketOptions: { timeout: 10000 } }); // إدارة الاتصال وتبادل الأوامر مع لوحة تحكم المتصفح (WebSockets) io.on('connection', (socket) => { // مزامنة فورية بمجرد تحميل الصفحة تضمن استجابة الشاشات والأزرار socket.emit('syncState', { isJoinOpen, users: Array.from(joinedUsers) }); socket.on('toggleJoin', (status) => { isJoinOpen = status; io.emit('joinStatusChanged', isJoinOpen); console.log(`[System]: وضع استقبال اللاعبين الآن: ${isJoinOpen ? 'مفتوح' : 'مغلق'}`); }); socket.on('adminAddUser', (name) => { if (name && !joinedUsers.has(name)) { joinedUsers.add(name); io.emit('updateUsers', Array.from(joinedUsers)); } }); socket.on('adminRemoveUser', (name) => { joinedUsers.delete(name); io.emit('updateUsers', Array.from(joinedUsers)); }); socket.on('adminClearAll', () => { joinedUsers.clear(); io.emit('updateUsers', Array.from(joinedUsers)); }); }); const processedMessages = new Set(); // الاستماع لتعليقات بث تيك توك المباشر في الوقت الفعلي tiktokConnection.on('chat', (data) => { if (!isJoinOpen) return; // تجاهل التعليقات إذا كان التسجيل مغلقاً في المتصفح if (data) { const msgId = data.msgId || (data.msg && data.msg.id); if (msgId && processedMessages.has(msgId)) return; if (msgId) processedMessages.add(msgId); const username = data.uniqueId || (data.user && (data.user.displayId || data.user.uniqueId)) || 'unknown'; const nickname = data.nickname || (data.user && data.user.nickname) || 'unknown'; const comment = (data.comment || data.text || data.content || '').trim(); // فحص الكلمة المفتاحية "انضم" if (comment === 'انضم' || comment.toLowerCase() === 'انضم') { if (!joinedUsers.has(nickname)) { joinedUsers.add(nickname); console.log(`➕ لاعب جديد انضم للبث: ${nickname} (@${username})`); io.emit('updateUsers', Array.from(joinedUsers)); } } } }); // إطلاق دالة الاتصال بالخلفية صامتة دون حظر الأزرار (Non-blocking) function runServer() { console.log('🔄 جاري فحص حالة البث في الخلفية دون حظر خادم الويب...'); tiktokConnection.waitUntilLive(30) .then(() => { console.log('🚀 الحساب نشط الآن! جاري بدء الاتصال...'); return tiktokConnection.connect(); }) .then(() => { console.log('✅ متصل بنجاح ببث تيك توك!'); }) .catch((err) => { console.error('❌ تنبيه في الخلفية (سيتم إعادة المحاولة تلقائياً):', err.toString()); setTimeout(runServer, 30000); }); } // تشغيل خادم الويب على المنفذ المتغير المعتمد لدى منصات الاستضافة (Render) const PORT = process.env.PORT || 3000; server.listen(PORT, () => { console.log(`🚀 السيرفر يعمل وجاهز ومستقر تماماً على المنفذ: ${PORT}`); runServer(); });
